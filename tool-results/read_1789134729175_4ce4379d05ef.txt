    16→function humanDuration(openedAt: Date): string {
    17→  const ms = Date.now() - openedAt.getTime()
    18→  const mins = Math.floor(ms / 60000)
    19→  if (mins < 60) return `${mins} мин`
    20→  const h = Math.floor(mins / 60)
    21→  const m = mins % 60
    22→  return `${h}ч ${m}м`
    23→}

   414→  // Хелпер: запустить AI с таймаутом 50 сек (чтобы успеть ответить до Telegram retry)
   415→  function withTimeout<T>(promise: Promise<T>, ms = 50000, label = 'AI'): Promise<T> {
   416→    return new Promise<T>((resolve, reject) => {
   417→      const timer = setTimeout(() => reject(new Error(`${label} timeout after ${ms}ms`)), ms)
   418→      promise.then(
   419→        (v) => { clearTimeout(timer); resolve(v) },
   420→        (e) => { clearTimeout(timer); reject(e) },
   421→      )
   422→    })
   423→  }

   425→  // Цикл "typing..." пока идёт долгая AI-обработка (каждые 4 сек, пока Telegram не сбросит)
   426→  async function keepTyping(ctx: { sendChatAction: (a: string) => Promise<unknown> }, stopFlag: { value: boolean }) {
   427→    while (!stopFlag.value) {
   428→      try { await ctx.sendChatAction('typing') } catch {}
   429→      await new Promise((r) => setTimeout(r, 4000))
   430→    }
   431→  }

   445→    if (text.startsWith('/')) return // команды уже обработаны
   446→
   447→    // Быстрый парсинг: "забил 5 кальянов", "сделал 3", "накрутил 10"
   448→    const { parseHookahCount } = await import('@/lib/datetime-utils')
   449→    const hookahCount = parseHookahCount(text)
   450→
   451→    // Если похоже на отчёт о кальянах и нет других ключевых слов AI
   452→    const lowerText = text.toLowerCase()
   453→    const hasOtherIntent = /сколько|чего|какой|какая|закажи|хочу|заканчивается|осталось|приход|накладн/.test(lowerText)
   454→
   455→    if (hookahCount !== null && hookahCount > 1 && !hasOtherIntent) {
   456→      // Пакетное добавление кальянов (минуя AI — быстро)
   457→      const shift = await db.shift.findFirst({ where: { masterId: master.id, status: 'OPEN' } })
   458→      if (!shift) {
   459→        return ctx.reply('⚠️ Смена не открыта. /shift чтобы открыть.')
   460→      }
   461→      const newCount = shift.hookahCount + hookahCount
   462→      await db.shift.update({ where: { id: shift.id }, data: { hookahCount: newCount } })
   463→      return ctx.reply(`🪔 +${hookahCount} кальянов\nВсего за смену: ${newCount}`)
   464→    }
   465→
   466→    await ctx.sendChatAction('typing')
   467→    const stop = { value: false }
   468→    const typingLoop = keepTyping(ctx, stop).catch(() => {})
   469→
   470→    try {
   471→      const result = await withTimeout(
   472→        processMasterMessage(text, { source: 'TEXT', masterId: master.id }),
   473→        50000,
   474→        'processMasterMessage',
   475→      )
   476→      stop.value = true
   477→      await typingLoop
   478→      return ctx.reply(formatReply(result))
   479→    } catch (e) {
   480→      stop.value = true
   481→      await typingLoop
   482→      console.error('Text/AI error:', (e as Error).message)
   483→      return ctx.reply('⚠️ Не удалось обработать (AI не ответил вовремя). Попробуй ещё раз.')
   484→    }
   485→  })

   487→  // Голосовые → ASR → AI
   488→  bot.on('voice', async (ctx) => {

   496→    }

   516→      }
   517→      if (!transcribedText) {
   518→        stop.value = true; await typingLoop

   520→      }
   521→

   523→      const { parseHookahCount } = await import('@/lib/datetime-utils')
   524→      const hookahCount = parseHookahCount(transcribedText)
   525→      const hasOtherIntent = /сколько|чего|какой|какая|закажи|хочу|заканчивается|осталось|приход|накладн/.test(transcribedText.toLowerCase())
   526→

   541→      )

   543→      return ctx.reply(`🎙 Распознал: «${transcribedText}»\n\n${formatReply(result)}`)
   544→    } catch (e) {

   560→    }

   567→      const photos = (ctx.message as { photo: { file_id: string }[] }).photo

   570→      const res = await fetch(fileLink.toString())
   571→      const buf = Buffer.from(await res.arrayBuffer())
   572→      const imageBase64 = 'data:image/jpeg;base64,' + buf.toString('base64')
   573→
   574→      const items = await withTimeout(recognizeInvoice(imageBase64), 30000, 'VLM')
   575→      if (items.length === 0) {
   576→        stop.value = true; await typingLoop
   577→        return ctx.reply('⚠️ Не удалось распознать позиции на фото.')
   578→      }
   579→
   580→      const result = await withTimeout(
   581→        processMasterMessage('оформи приход по накладной', { source: 'PHOTO', invoiceItems: items, masterId: master.id }),
   582→        50000,
   583→        'processMasterMessage',
   584→      )
   585→
   586→      let reply = `📸 Распознал ${items.length} поз.:\n`
   587→      for (const i of items) reply += `• ${i.brand} ${i.line} ${i.flavor} — ${i.grams}г\n`
   588→      reply += `\n${formatReply(result)}`
   589→      stop.value = true; await typingLoop
   590→      return ctx.reply(reply)

   592→      stop.value = true; await typingLoop

   594→      return ctx.reply('⚠️ Не удалось обработать фото (таймаут или ошибка AI).')
   595→    }
   596→  })

   603→  const shift = await db.shift.findFirst({ where: { masterId, status: 'OPEN' } })
   604→  if (!shift) {

   606→  }
   607→  const newCount = shift.hookahCount + 1
   608→  await db.shift.update({ where: { id: shift.id }, data: { hookahCount: newCount } })

   612→function formatReply(result: {
   613→  reply: string
   614→  executedActions?: Array<{ tool: string; success: boolean; message: string }>
   615→}): string {
   616→  let reply = result.reply || 'Готово.'
   617→  if (result.executedActions && result.executedActions.length > 0) {
   618→    reply += '\n\n'
   619→    for (const a of result.executedActions) {
   620→      reply += `${a.success ? '✅' : '❌'} ${a.message}\n`
   621→    }
   622→  }
   623→  return reply.trim()
   624→}

   634→export async function ensureBotRunning(): Promise<void> {
   635→  if (globalForBot.__botStarted) return
   636→  globalForBot.__botStarted = true
   637→
   638→  try {
   639→    const bot = createBot()
   640→    globalForBot.__botInstance = bot
   641→
   642→    // НЕ блокируем рендер страницы — getMe и launch в фоне
   643→    void (async () => {
   644→      try {
   645→        const me = await bot.telegram.getMe()
   646→        console.log(`🤖 Telegram бот инициализирован: @${me.username}`)
   647→
   648→        if (USE_POLLING) {
   649→          bot.launch()
   650→          console.log('✅ Polling активен (USE_POLLING=1).')
   651→          process.once('SIGINT', () => bot.stop('SIGINT'))
   652→          process.once('SIGTERM', () => bot.stop('SIGTERM'))
   653→        } else {
   654→          console.log('✅ Webhook режим. Ожидаем Update на /api/telegram/webhook')
   655→        }
   656→      } catch (e) {
   657→        console.error('⚠️ Фоновая инициализация бота не удалась:', (e as Error).message)
   658→      }
   659→    })()
   660→
   661→    console.log('🔄 Бот инициализируется в фоне (не блокирует запросы)')
   662→  } catch (e) {
   663→    console.error('❌ Ошибка инициализации бота:', (e as Error).message)
   664→    globalForBot.__botStarted = false
   665→  }
   666→}
