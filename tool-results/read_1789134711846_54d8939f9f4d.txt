   180→   180→// Fuzzy поиск табака в базе по названию
   181→   181→async function findTobacco(brand: string, line: string, flavor: string) {
   182→   182→  const all = await db.tobacco.findMany({ where: { active: true }, include: { stock: true } })
   183→   183→
   184→   184→  const normalize = (s: string) => s.toLowerCase().trim()
   185→   185→  const b = normalize(brand)
   186→   186→  const l = normalize(line)
   187→   187→  const f = normalize(flavor)
   188→   188→
   189→   189→  let match = all.find(
   190→   190→    (t) => normalize(t.brand) === b && normalize(t.line) === l && normalize(t.flavor) === f,
   191→   191→  )
   192→   192→  if (match) return match
   193→   193→
   194→   194→  match = all.find((t) => normalize(t.brand) === b && normalize(t.flavor) === f)
   195→   195→  if (match) return match
   196→   196→
   197→   197→  if (!flavor) {
   198→   198→    match = all.find((t) => normalize(t.brand) === b && normalize(t.line) === l)
   199→   199→    if (match) return match
   200→   200→  }
   201→   201→
   202→   202→  match = all.find(
   203→   203→    (t) =>
   204→   204→      normalize(t.brand).includes(b) ||
   205→   205→      (b && b.includes(normalize(t.brand))) ||
   206→   206→      normalize(t.flavor).includes(f) ||
   207→   207→      (f && f.includes(normalize(t.flavor))),
   208→   208→  )
   209→   209→  return match
   210→   210→}

   252→   252→  // Недавние операции за текущие смены (что мастера отметили)

   255→   255→    const shiftMasterIds = openShifts.map((s) => s.masterId)

   270→   270→    for (const op of ops) {

   272→   272→      const t = op.tobacco

   274→   274→    }

   284→   284→      const minsAgo = Math.floor((Date.now() - s.openedAt.getTime()) / 60000)
   285→   285→      const dur = hoursAgo > 0 ? `${hoursAgo}ч` : `${minsAgo}м`
   286→   286→      const mine = s.masterId === master.id ? ' (ты)' : ''
   287→   287→      lines.push(`- ${s.master.name}${mine} | на смене ${dur} | ${s.hookahCount} кальянов`)
   288→   288→    }
   289→   289→  }
   290→   290→
   291→   291→  if (recentOps.length > 0) {
   292→   292→    lines.push('\nНЕДАВНИЕ ОПЕРАЦИИ ПО ОСТАТКАМ (за смену):')
   293→   293→    lines.push(...recentOps)
   294→   294→  }
   295→   295→
   296→   296→  if (activeRequests.length > 0) {
   297→   297→    lines.push('\nАКТИВНЫЕ ЗАЯВКИ МАСТЕРОВ:')
   298→   298→    for (const r of activeRequests) {
   299→   299→      lines.push(`- ${r.master.name}: "${r.text}"${r.grams ? ` (${r.grams}г)` : ''}`)
   300→   300→    }
   301→   301→  }
   302→   302→
   303→   303→  if (pendingWishes.length > 0) {
   304→   304→    lines.push('\nХОТЕЛКИ МАСТЕРОВ:')
   305→   305→    for (const w of pendingWishes) {
   306→   306→      lines.push(`- ${w.master.name}: "${w.text}"`)
   307→   307→    }
   308→   308→  }
   309→   309→
   310→   310→  return lines.join('\n')
   311→   311→}

   313→   313→// Парсинг JSON из ответа LLM (с защитой от markdown обёртки и reasoning)
   314→   314→function parseAIResponse(content: string): { actions: AIAction[]; reply: string } {
   315→   315→  let cleaned = content.trim()
   316→   316→
   317→   317→  // Убираем markdown обёртку ```json ... ```
   318→   318→  const jsonMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/)
   319→   319→  if (jsonMatch) {
   320→   320→    cleaned = jsonMatch[1].trim()
   321→   321→  }
   322→   322→
   323→   323→  // Ищем JSON объект в тексте (модели иногда добавляют рассуждения до/после)
   324→   324→  const jsonObj = cleaned.match(/\{[\s\S]*\}/)
   325→   325→  if (jsonObj) {
   326→   326→    cleaned = jsonObj[0]
   327→   327→  }
   328→   328→
   329→   329→  try {
   330→   330→    const parsed = JSON.parse(cleaned)
   331→   331→    return {
   332→   332→      actions: Array.isArray(parsed.actions) ? parsed.actions : [],
   333→   333→      reply: typeof parsed.reply === 'string' ? parsed.reply : 'Готово.',
   334→   334→    }
   335→   335→  } catch {
   336→   336→    return { actions: [], reply: content }
   337→   337→  }
   338→   338→}

   340→   340→// Выполнение одного action
   341→   341→async function executeAction(action: AIAction, master: SessionMaster): Promise<{ success: boolean; message: string; data?: unknown }> {
   342→   342→  try {
   343→   343→    switch (action.tool) {
   344→   344→      case 'update_stock': {
   345→   345→        const { brand, line, flavor, grams, note } = action.args
   346→   346→        const tobacco = await findTobacco(brand, line, flavor)
   347→   347→        if (!tobacco) {
   348→   348→          return { success: false, message: `Табак "${brand} ${line} ${flavor}" не найден в базе` }
   349→   349→        }

   386→   386→
   387→   387→      case 'add_incoming': {

   402→   402→            tobaccoId: tobacco.id,
   403→   403→            type: 'INCOMING',
   404→   404→            gramsBefore: before,
   405→   405→            gramsAfter: after,
   406→   406→            delta: grams,
   407→   407→            source: 'TEXT',
   408→   408→            note: note ?? 'Приход',

   412→   412→          success: true,

   423→   423→        }

   425→   425→          data: { brand, line, flavor, defaultJarGrams, thresholdGrams: 70 },

   462→   462→        }

   464→   464→      }
   465→   465→

   478→   478→
   479→   479→      case 'add_hookah_batch': {
   480→   480→        const count = Math.max(1, Math.min(100, action.args.count))
   481→   481→        const shift = await db.shift.findFirst({ where: { masterId: master.id, status: 'OPEN' } })
   482→   482→        if (!shift) {

   484→   484→        }

   492→   492→      }
   493→   493→
   494→   494→      case 'query': {
   495→   495→        const { what } = action.args
   496→   496→        const w = String(what).toLowerCase().replace(/[\s_-]+/g, '')

   503→   503→        }
   504→   504→        if (w.includes('all') || w.includes('stock') || w.includes('full')) {
   505→   505→          const all = await db.tobacco.findMany({ where: { active: true }, include: { stock: true }, orderBy: [{ brand: 'asc' }, { flavor: 'asc' }] })
   506→   506→          return { success: true, message: `Всего ${all.length} позиций`, data: all.map((t) => ({ brand: t.brand, line: t.line, flavor: t.flavor, current: t.stock?.currentGrams ?? 0 })) }
   507→   507→        }
   508→   508→        if (w.includes('shift')) {
   509→   509→          const shiftContext = await buildShiftContext(master)
   510→   510→          return { success: true, message: 'Контекст смены загружен', data: { shiftContext } }
   511→   511→        }
   512→   512→        const low = await db.tobacco.findMany({ where: { active: true }, include: { stock: true } })
   513→   513→        const filtered = low
   514→   514→          .filter((t) => (t.stock?.currentGrams ?? 0) < t.thresholdGrams)
   515→   515→          .map((t) => ({ brand: t.brand, line: t.line, flavor: t.flavor, current: t.stock?.currentGrams ?? 0, threshold: t.thresholdGrams }))
   516→   516→        return { success: true, message: `Показано ${filtered.length} позиций "мало" (fallback)`, data: filtered }
   517→   517→      }
   518→   518→
   519→   519→      default:
   520→   520→        return { success: false, message: 'Неизвестное действие' }
   521→   521→    }
   522→   522→  } catch (e) {
   523→   523→    return { success: false, message: `Ошибка: ${(e as Error).message}` }
   524→   524→  }

   605→   605→export async function recognizeInvoice(imageBase64: string): Promise<Array<{ brand: string; line: string; flavor: string; grams: number }>> {
   606→   606→  const prompt = `Ты распознаёшь накладную на кальянный табак. Найди ВСЕ позиции табака на изображении.
   607→   607→Для каждой позиции верни: brand (бренд/производитель), line (линейка, если есть), flavor (вкус), grams (вес в граммах одной банки/упаковки).
   608→   608→Если вес указан в граммах — верни число. Если в банках/штуках — верни вес одной банки.
   609→   609→
   610→   610→Верни СТРОГО JSON массив без markdown:
   611→   611→[
   612→   612→  { "brand": "Darkside", "line": "Supernova", "flavor": "Ice Grape", "grams": 250 }
   613→   613→]`
   614→   614→
   615→   615→  const isDataUrl = imageBase64.startsWith('data:')
   616→   616→  const base64Data = isDataUrl ? imageBase64.split(',')[1] : imageBase64
   617→   617→  const mimeType = imageBase64.match(/data:(image\/[\w+]+);/)?.[1] || 'image/jpeg'
   618→   618→
   619→   619→  const content = await hfChat(
   620→   620→    [
   621→   621→      { role: 'user', content: [
   622→   622→        { type: 'text', text: prompt },
   623→   623→        { type: 'image_url', image_url: { url: `data:${mimeType};base64,${base64Data}` } },
   624→   624→      ] },
   625→   625→    ],
   626→   626→    { vision: true, maxTokens: 1000 },
   627→   627→  )
   628→   628→
   629→   629→  let cleaned = content.trim()
   630→   630→  const jsonMatch = cleaned.match(/\[[\s\S]*\]/)
   631→   631→  if (jsonMatch) cleaned = jsonMatch[0]
   632→   632→
   633→   633→  try {
   634→   634→    const parsed = JSON.parse(cleaned)
   635→   635→    if (Array.isArray(parsed)) {
   636→   636→      return parsed.filter(
   637→   637→        (i: { brand?: string; flavor?: string; grams?: number }) => i.brand && i.flavor && i.grams,
   638→   638→      )
   639→   639→    }
   640→   640→    return []
   641→   641→  } catch {
   642→   642→    return []
   643→   643→  }
   644→   644→}

   647→   647→// Транскрипция голоса через HF Whisper Large V3 Turbo (бесплатно)
   648→   648→// ───────────────────────────────────────────
   649→   649→const WHISPER_API = 'https://router.huggingface.co/hf-inference/models/openai/whisper-large-v3-turbo'
   650→   650→
   651→   651→export async function transcribeAudio(audioBase64: string): Promise<string> {
   652→   652→  const token = getHfToken()
   653→   653→
   654→   654→  // Определяем формат аудио (Telegram присылает OGG/Opus)
   655→   655→  let mimeType = 'audio/wav'
   656→   656→  if (audioBase64.startsWith('data:')) {
   657→   657→    const match = audioBase64.match(/^data:(audio\/[\w+.-]+);/)
   658→   658→    if (match) {
   659→   659→      mimeType = match[1]
   660→   660→      audioBase64 = audioBase64.split(',')[1]
   661→   661→    }
   662→   662→  } else {
   663→   663→    // Telegram OGG начинается с байтов 'OggS'
   664→   664→    try {
   665→   665→      const head = Buffer.from(audioBase64.slice(0, 8), 'base64').toString('ascii')
   666→   666→      if (head.startsWith('OggS')) mimeType = 'audio/ogg'
   667→   667→    } catch {}
   668→   668→  }
   669→   669→
   670→   670→  const audioBuffer = Buffer.from(audioBase64, 'base64')
   671→   671→
   672→   672→  const res = await fetch(WHISPER_API, {
   673→   673→    method: 'POST',
   674→   674→    headers: {
   675→   675→      Authorization: `Bearer ${token}`,
   676→   676→      'Content-Type': mimeType,
   677→   677→    },
   678→   678→    body: audioBuffer,
   679→   679→  })
   680→   680→
   681→   681→  const data = (await res.json()) as { text?: string; error?: { message?: string } }
   682→   682→  if (!res.ok || data.error) {
   683→   683→    throw new Error(`Whisper: ${data.error?.message || res.status}`)
   684→   684→  }
   685→   685→  return (data.text || '').trim()
   686→   686→}
