    30→  choices?: Array<{

    37→async function hfChat(messages: HFChatMessage[], opts: { vision?: boolean; maxTokens?: number } = {}): Promise<string> {
    38→  const model = opts.vision ? VISION_MODEL : LLM_MODEL
    39→  const provider = opts.vision ? VISION_PROVIDER : LLM_PROVIDER
    40→
    41→  const body: Record<string, unknown> = {
    42→    model,
    43→    provider,
    44→    messages,
    45→    max_tokens: opts.maxTokens ?? 1500,
    46→  }
    47→
    48→  const res = await fetch(HF_API, {
    49→    method: 'POST',
    50→    headers: {
    51→      Authorization: `Bearer ${getHfToken()}`,
    52→      'Content-Type': 'application/json',
    53→    },
    54→    body: JSON.stringify(body),
    55→  })
    56→
    57→  const data = (await res.json()) as HFChatResponse
    58→  if (!res.ok || data.error) {
    59→    throw new Error(data.error?.message || `HF API ${res.status}`)
    60→  }
    61→
    62→  const content = data.choices?.[0]?.message?.content ?? ''
    63→  // Qwen3.8 возвращает reasoning вместо content для reasoning-моделей
    64→  if (!content && data.choices?.[0]?.message?.reasoning) {
    65→    return data.choices[0].message.reasoning
    66→  }
    67→  return content
    68→}

   180→// Fuzzy поиск табака в базе по названию
   181→async function findTobacco(brand: string, line: string, flavor: string) {
   182→  const all = await db.tobacco.findMany({ where: { active: true }, include: { stock: true } })
   183→
   184→  const normalize = (s: string) => s.toLowerCase().trim()
   185→  const b = normalize(brand)
   186→  const l = normalize(line)
   187→  const f = normalize(flavor)
   188→
   189→  let match = all.find(
   190→    (t) => normalize(t.brand) === b && normalize(t.line) === l && normalize(t.flavor) === f,
   191→  )
   192→  if (match) return match
   193→
   194→  match = all.find((t) => normalize(t.brand) === b && normalize(t.flavor) === f)
   195→  if (match) return match
   196→
   197→  if (!flavor) {
   198→    match = all.find((t) => normalize(t.brand) === b && normalize(t.line) === l)
   199→    if (match) return match
   200→  }
   201→
   202→  match = all.find(
   203→    (t) =>
   204→      normalize(t.brand).includes(b) ||
   205→      (b && b.includes(normalize(t.brand))) ||
   206→      normalize(t.flavor).includes(f) ||
   207→      (f && f.includes(normalize(t.flavor))),
   208→  )
   209→  return match
   210→}

   212→// Контекст склада для промпта
   213→async function buildStockContext(): Promise<string> {
   214→  const tobaccos = await db.tobacco.findMany({
   215→    where: { active: true },
   216→    include: { stock: true },
   217→    orderBy: [{ brand: 'asc' }, { line: 'asc' }, { flavor: 'asc' }],
   218→  })
   219→
   220→  if (tobaccos.length === 0) return '(справочник пуст)'
   221→
   222→  const lines = tobaccos.map((t) => {
   223→    const grams = t.stock?.currentGrams ?? 0
   224→    const status = grams < t.thresholdGrams ? ' ⚠️ МАЛО' : ''
   225→    return `- ${t.brand} / ${t.line} / ${t.flavor} | банка=${t.defaultJarGrams}г | остаток=${grams}г | порог=${t.thresholdGrams}г${status}`
   226→  })
   227→  return lines.join('\n')
   228→}

   230→// Контекст смены для промпта
   231→async function buildShiftContext(master: SessionMaster): Promise<string> {
   232→  const openShifts = await db.shift.findMany({
   233→    where: { status: 'OPEN' },
   234→    include: { master: true },
   235→    orderBy: { openedAt: 'asc' },
   236→  })
   237→
   238→  const activeRequests = await db.masterRequest.findMany({
   239→    where: { status: 'PENDING' },
   240→    include: { master: true },
   241→    orderBy: { createdAt: 'desc' },
   242→    take: 10,
   243→  })
   244→
   245→  const pendingWishes = await db.wish.findMany({
   246→    where: { status: 'PENDING' },
   247→    include: { master: true },
   248→    orderBy: { createdAt: 'desc' },
   249→    take: 10,
   250→  })
   251→
   252→  // Недавние операции за текущие смены (что мастера отметили)
   253→  const recentOps: string[] = []
   254→  if (openShifts.length > 0) {
   255→    const shiftMasterIds = openShifts.map((s) => s.masterId)
   256→    const earliestOpen = openShifts.reduce((min, s) => (s.openedAt < min ? s.openedAt : min), openShifts[0].openedAt)
   257→
   258→    const ops = await db.operation.findMany({
   259→      where: {
   260→        createdAt: { gte: earliestOpen },
   261→        type: 'ADJUSTMENT',
   262→      },
   263→      include: { tobacco: true },
   264→      orderBy: { createdAt: 'desc' },
   265→      take: 15,
   266→    })
   267→
   268→    // Группируем по мастерам (но operation не имеет masterId напрямую — берём из ChatMessage)
   269→    // Проще: показать все недавние операции с заметками
   270→    for (const op of ops) {
   271→      const time = new Date(op.createdAt).toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' })
   272→      const t = op.tobacco
   273→      recentOps.push(`  [${time}] ${t.brand} ${t.line} ${t.flavor}: ${op.gramsBefore}г → ${op.gramsAfter}г${op.note ? ` (${op.note})` : ''}`)
   274→    }
   275→  }
   276→
   277→  const lines: string[] = []
   278→  lines.push('ТЕКУЩАЯ СМЕНА:')
   279→  if (openShifts.length === 0) {
   280→    lines.push('(никого на смене)')
   281→  } else {
   282→    for (const s of openShifts) {
   283→      const hoursAgo = Math.floor((Date.now() - s.openedAt.getTime()) / 3600000)
   284→      const minsAgo = Math.floor((Date.now() - s.openedAt.getTime()) / 60000)
   285→      const dur = hoursAgo > 0 ? `${hoursAgo}ч` : `${minsAgo}м`
   286→      const mine = s.masterId === master.id ? ' (ты)' : ''
   287→      lines.push(`- ${s.master.name}${mine} | на смене ${dur} | ${s.hookahCount} кальянов`)
   288→    }
   289→  }
   290→
   291→  if (recentOps.length > 0) {
   292→    lines.push('\nНЕДАВНИЕ ОПЕРАЦИИ ПО ОСТАТКАМ (за смену):')
   293→    lines.push(...recentOps)
   294→  }
   295→
   296→  if (activeRequests.length > 0) {
   297→    lines.push('\nАКТИВНЫЕ ЗАЯВКИ МАСТЕРОВ:')
   298→    for (const r of activeRequests) {
   299→      lines.push(`- ${r.master.name}: "${r.text}"${r.grams ? ` (${r.grams}г)` : ''}`)
   300→    }
   301→  }
   302→
   303→  if (pendingWishes.length > 0) {
   304→    lines.push('\nХОТЕЛКИ МАСТЕРОВ:')
   305→    for (const w of pendingWishes) {
   306→      lines.push(`- ${w.master.name}: "${w.text}"`)
   307→    }
   308→  }
   309→
   310→  return lines.join('\n')
   311→}

   313→// Парсинг JSON из ответа LLM (с защитой от markdown обёртки и reasoning)
   314→function parseAIResponse(content: string): { actions: AIAction[]; reply: string } {
   315→  let cleaned = content.trim()
   316→
   317→  // Убираем markdown обёртку ```json ... ```
   318→  const jsonMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/)
   319→  if (jsonMatch) {
   320→    cleaned = jsonMatch[1].trim()
   321→  }
   322→
   323→  // Ищем JSON объект в тексте (модели иногда добавляют рассуждения до/после)
   324→  const jsonObj = cleaned.match(/\{[\s\S]*\}/)
   325→  if (jsonObj) {
   326→    cleaned = jsonObj[0]
   327→  }
   328→
   329→  try {
   330→    const parsed = JSON.parse(cleaned)
   331→    return {
   332→      actions: Array.isArray(parsed.actions) ? parsed.actions : [],
   333→      reply: typeof parsed.reply === 'string' ? parsed.reply : 'Готово.',
   334→    }
   335→  } catch {
   336→    return { actions: [], reply: content }
   337→  }
   338→}

   340→// Выполнение одного action
   341→async function executeAction(action: AIAction, master: SessionMaster): Promise<{ success: boolean; message: string; data?: unknown }> {
   342→  try {
   343→    switch (action.tool) {
   344→      case 'update_stock': {
   345→        const { brand, line, flavor, grams, note } = action.args
   346→        const tobacco = await findTobacco(brand, line, flavor)
   347→        if (!tobacco) {
   348→          return { success: false, message: `Табак "${brand} ${line} ${flavor}" не найден в базе` }
   349→        }
   350→        const before = tobacco.stock?.currentGrams ?? 0
   351→        await db.stockItem.upsert({
   352→          where: { tobaccoId: tobacco.id },
   353→          update: { currentGrams: grams },
   354→          create: { tobaccoId: tobacco.id, currentGrams: grams },
   355→        })
   356→        await db.operation.create({
   357→          data: {
   358→            tobaccoId: tobacco.id,
   359→            type: 'ADJUSTMENT',
   360→            gramsBefore: before,
   361→            gramsAfter: grams,
   362→            delta: grams - before,
   363→            source: 'TEXT',
   364→            note: note ?? null,
   365→          },
   366→        })
   367→        if (master.role !== 'SENIOR' && (grams === 0 || grams < tobacco.thresholdGrams)) {
   368→          const reason = grams === 0 ? 'закончился' : `мало осталось (${grams}г)`
   369→          const notifMsg = `⚠️ ${master.name} отметил: ${reason} ${tobacco.brand} ${tobacco.line} ${tobacco.flavor}`
   370→          await db.notification.create({
   371→            data: {
   372→              type: 'FINISHED',
   373→              message: notifMsg,
   374→              masterId: master.id,
   375→            },
   376→          })
   377→          // Push в Telegram старшему
   378→          await pushToSeniors(notifMsg)
   379→        }
   380→        return {
   381→          success: true,
   382→          message: `${tobacco.brand} ${tobacco.line} ${tobacco.flavor}: ${before}г → ${grams}г`,
   383→          data: { before, after: grams, tobaccoId: tobacco.id },
   384→        }
   385→      }
   386→
   387→      case 'add_incoming': {
   388→        const { brand, line, flavor, grams, note } = action.args
   389→        const tobacco = await findTobacco(brand, line, flavor)
   390→        if (!tobacco) {
   391→          return { success: false, message: `Табак "${brand} ${line} ${flavor}" не найден в базе` }
   392→        }
   393→        const before = tobacco.stock?.currentGrams ?? 0
   394→        const after = before + grams
   395→        await db.stockItem.upsert({
   396→          where: { tobaccoId: tobacco.id },
   397→          update: { currentGrams: after },
   398→          create: { tobaccoId: tobacco.id, currentGrams: after },
   399→        })
   400→        await db.operation.create({
   401→          data: {
   402→            tobaccoId: tobacco.id,
   403→            type: 'INCOMING',
   404→            gramsBefore: before,
   405→            gramsAfter: after,
   406→            delta: grams,
   407→            source: 'TEXT',
   408→            note: note ?? 'Приход',
   409→          },
   410→        })
   411→        return {
   412→          success: true,
   413→          message: `Приход ${tobacco.brand} ${tobacco.flavor}: +${grams}г (стало ${after}г)`,
   414→          data: { before, after, added: grams, tobaccoId: tobacco.id },
   415→        }
   416→      }
   417→
   418→      case 'add_tobacco': {
   419→        const { brand, line, flavor, defaultJarGrams = 250 } = action.args

   651→export async function transcribeAudio(audioBase64: string): Promise<string> {
   652→  const token = getHfToken()
   653→
   654→  // Определяем формат аудио (Telegram присылает OGG/Opus)
   655→  let mimeType = 'audio/wav'
   656→  if (audioBase64.startsWith('data:')) {
   657→    const match = audioBase64.match(/^data:(audio\/[\w+.-]+);/)
   658→    if (match) {
   659→      mimeType = match[1]
   660→      audioBase64 = audioBase64.split(',')[1]
   661→    }
   662→  } else {
   663→    // Telegram OGG начинается с байтов 'OggS'
   664→    try {
   665→      const head = Buffer.from(audioBase64.slice(0, 8), 'base64').toString('ascii')
   666→      if (head.startsWith('OggS')) mimeType = 'audio/ogg'
   667→    } catch {}
   668→  }
   669→
   670→  const audioBuffer = Buffer.from(audioBase64, 'base64')
   671→
   672→  const res = await fetch(WHISPER_API, {
   673→    method: 'POST',
   674→    headers: {
   675→      Authorization: `Bearer ${token}`,
   676→      'Content-Type': mimeType,
   677→    },
   678→    body: audioBuffer,
   679→  })
   680→
   681→  const data = (await res.json()) as { text?: string; error?: { message?: string } }
   682→  if (!res.ok || data.error) {
   683→    throw new Error(`Whisper: ${data.error?.message || res.status}`)
   684→  }
   685→  return (data.text || '').trim()
   686→}
